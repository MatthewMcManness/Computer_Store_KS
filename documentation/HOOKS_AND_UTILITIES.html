useBotProtection

File: src/hooks/useBotProtection.ts

Purpose: Provides timing analysis and honeypot fields to detect automated form submissions. This is a lightweight protection mechanism that should be combined with server-side validation.

Parameters: None

Returns:

Property Type Description
timing string Unix timestamp (milliseconds) of when the component loaded
honeypotFields object Hidden field values that should remain empty
honeypotFields._hp_email2 string Empty honeypot field
honeypotFields._hp_phone_confirm string Empty honeypot field
honeypotFields._hp_url string Empty honeypot field

Usage Example:

const ContactForm = () => {
  const { timing, honeypotFields } = useBotProtection();

  const handleSubmit = (e) => {
    const formData = {
      ...regularFields,
      pageLoadTime: timing,
      submitTime: Date.now().toString(),
      ...honeypotFields
    };
    // Server validates timing and ensures honeypot fields are empty
    submitForm(formData);
  };

  return (
    <form onSubmit={handleSubmit}>
      {/* Hidden honeypot fields - invisible to users */}
      <input type="text" name="_hp_email2" value={honeypotFields._hp_email2}
             style={{ display: 'none' }} tabIndex={-1} autoComplete="off" />
      {/* Regular form fields */}
    </form>
  );
};

Status: Production

useFingerprint

File: src/hooks/useFingerprint.ts

Purpose: Generates advanced browser fingerprints using FingerprintJS library. Creates unique visitor IDs based on dozens of browser and device characteristics for bot detection.

Parameters: None

Returns:

Property Type Description
fingerprint FingerprintData Object containing fingerprint results
fingerprint.visitorId string Unique visitor ID generated by FingerprintJS
fingerprint.confidence number Confidence score (0-1)
fingerprint.success boolean Whether fingerprinting was successful
fingerprint.error string? Error message if fingerprinting failed
isLoading boolean Whether fingerprinting is in progress
getFingerprintSpamScore () => number Function returning spam score (0-10) based on fingerprint quality

Fingerprint Signals Include:

  • Canvas rendering fingerprint
  • WebGL rendering capabilities
  • Audio context properties
  • Installed system fonts
  • Hardware concurrency (CPU cores)
  • Device memory
  • Screen resolution and color depth
  • Timezone and language settings
  • Browser plugins and features
  • And 40+ other signals

Usage Example:

const ContactForm = () => {
  const { fingerprint, isLoading, getFingerprintSpamScore } = useFingerprint();

  const handleSubmit = () => {
    if (isLoading) return;

    const formData = {
      ...fields,
      _fingerprint: {
        visitorId: fingerprint.visitorId,
        confidence: fingerprint.confidence,
        simpleFingerprint: getSimpleFingerprint(),
        spamScore: getFingerprintSpamScore()
      }
    };
    submitForm(formData);
  };
};

Status: Production

useInteractionTracking

File: src/hooks/useInteractionTracking.ts

Purpose: Tracks user interactions to distinguish humans from bots. Monitors various interaction types and timing patterns to build a behavioral profile.

Tracked Interactions:

  • Mouse movements (throttled to max 10/second)
  • Clicks
  • Keystrokes
  • Scroll events
  • Focus events
  • Timing variance between interactions

Parameters: None

Returns:

Property Type Description
getInteractionScore () => InteractionScore Function returning detailed scoring analysis
getInteractionData () => InteractionData Function returning raw interaction counts

InteractionScore Type:

interface InteractionScore {
  score: number;        // 0-6 (number of human-like behaviors detected)
  maxScore: number;     // Always 6
  isHumanLike: boolean; // true if score >= 3
  spamScore: number;    // 0-20 (inverse of human score)
  details: {
    hasMouseMovement: boolean;
    hasClicks: boolean;
    hasKeystrokes: boolean;
    hasScrolls: boolean;
    hasFocusEvents: boolean;
    hasNaturalVariance: boolean;
  };
}

Spam Score Calculation:

  • 0-2 human indicators = 20 spam points (likely bot)
  • 3-4 human indicators = 10 spam points (suspicious)
  • 5-6 human indicators = 0 spam points (human-like)

Usage Example:

const ContactForm = () => {
  const { getInteractionScore } = useInteractionTracking();

  const handleSubmit = () => {
    const { isHumanLike, spamScore, details } = getInteractionScore();

    if (!isHumanLike) {
      console.log('Bot detected', details);
      return;
    }

    const formData = {
      ...fields,
      _interaction: { ...getInteractionScore() }
    };
    submitForm(formData);
  };
};

Status: Production

useDarkMode

File: src/hooks/useDarkMode.ts

Purpose: Manages dark mode theme state in the admin dashboard with automatic persistence to localStorage and system preference detection.

Parameters: None

Returns:

Property Type Description
isDark boolean Whether dark mode is currently active
toggle () => void Function to toggle between light and dark mode
isLoaded boolean Whether preference has been loaded (prevents flashing)

Behavior:

  1. On initial load, checks localStorage for saved preference (key: admin-dark-mode)
  2. If no saved preference, falls back to system preference (prefers-color-scheme)
  3. Adds/removes dark class on document.documentElement for Tailwind CSS

Usage Example:

const AdminDashboard = () => {
  const { isDark, toggle, isLoaded } = useDarkMode();

  // Prevent flash of unstyled content
  if (!isLoaded) return <Spinner />;

  return (
    <div className="min-h-screen bg-white dark:bg-gray-900">
      <button onClick={toggle}>
        {isDark ? 'Light Mode' : 'Dark Mode'}
      </button>
    </div>
  );
};

Status: Production

Usage Patterns

Complete Bot Protection for Forms

'use client';

import { useBotProtection } from '@/hooks/useBotProtection';
import { useFingerprint, getSimpleFingerprint } from '@/hooks/useFingerprint';
import { useInteractionTracking } from '@/hooks/useInteractionTracking';

export function ContactForm() {
  const { timing, honeypotFields } = useBotProtection();
  const { fingerprint, isLoading, getFingerprintSpamScore } = useFingerprint();
  const { getInteractionScore } = useInteractionTracking();

  const handleSubmit = async (e: FormEvent) => {
    e.preventDefault();

    // Collect all bot protection data
    const formData = {
      name,
      email,
      message,
      // Timing
      pageLoadTime: parseInt(timing),
      submitTime: Date.now(),
      // Honeypots
      ...honeypotFields,
      // Fingerprint
      _fingerprint: {
        visitorId: fingerprint.visitorId,
        confidence: fingerprint.confidence,
        simpleFingerprint: getSimpleFingerprint(),
        spamScore: getFingerprintSpamScore()
      },
      // Interaction
      _interaction: getInteractionScore()
    };

    const response = await fetch('/api/contact', {

Server-Side Spam Detection

// app/api/contact/route.ts
import { calculateSpamScore, ExtendedFormData } from '@/lib/spam-detection';
import { sendContactNotification } from '@/lib/email';

export async function POST(request: Request) {
  const data: ExtendedFormData = await request.json();
  const headers = request.headers;
  const clientIP = headers.get('x-forwarded-for') || 'unknown';

  // Calculate spam score
  const result = await calculateSpamScore(data, headers, clientIP);

  // Handle based on action
  switch (result.action) {
    case 'silent_success':
      // Return fake success to bot
      return Response.json({ success: true });

    case 'block':
      return Response.json(
        { error: 'Message blocked as spam' },
        { status: 400 }
      );

    case 'log':
      // Log suspicious but allow
      console.log('Suspicious submission:', result.breakdown);
      // Fall through to allow

    case 'allow':
      await sendContactNotification(data);
      return Response.json({ success: true });
  }
}
